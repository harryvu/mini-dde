Week 6 · Part III — the delivery machine

Gates & enforcement

Process that lives in prose gets skipped under pressure — always, by humans and agents alike. This week you make your pipeline mechanically honest: gates are named condition sets a Responsibility must pass to advance; evidence is submitted and validated, never asserted; reviewer verdicts gate phases; and Claude Code hooks make "skip the gate" structurally impossible rather than merely forbidden.

  • Concept: a gate protects an exit, and evidence beats assertion
  • Build: check_gate / submit_gate_evidence / update_dod
  • Build: record_verdict + the reviewer ensemble
  • Build: exit-2 hooks + a sectioned pre-commit
6.1Gates: named exit criteriacheck_gate▶

A gate is a named list of conditions — some computed from the graph (all_findings_terminal), some requiring submitted evidence (tests_pass). check_gate evaluates every condition and passes only when zero required checks fail. The real system seeds 22 gate definitions at every server start from an in-code canon; your mini keeps three, which is enough to gate an entire pipeline.

In the real system

DefaultGateDefinitions() in baseline.go:116-141, seeded idempotently into the graph so projects can customize. The big three: design_F_exit (WPs registered, handoff exists, findings terminal, SDD completeness), build_phase_G (WPs complete, findings terminal, no internal deferrals, DoD complete, review docs have content, evidence-backed tests/static-analysis/deployment), acceptance_exit (scope-aware category matrix, fails closed on an unclassified scope). Evidence freshness is checked — a status regression or a source-file edit after submission invalidates it.

  1. Define gates and conditions in code:

    package graph

import (
	"context"
	"fmt"
)

type GateCondition struct {
	Check            string `json:"check"`
	Required         bool   `json:"required"`
	EvidenceRequired bool   `json:"evidence_required"`
}

type GateResult struct {
	Gate    string            `json:"gate"`
	Passed  bool              `json:"passed"`
	Results map[string]string `json:"results"` // check → "pass" | reason
}

// The mini canon. The real server seeds 22 of these into the graph.
var GateDefinitions = map[string][]GateCondition{
	"design_exit": {
		{Check: "has_specification", Required: true},
		{Check: "all_findings_terminal", Required: true},
		{Check: "work_packages_registered", Required: true},
	},
	"build_exit": { // the mini build_phase_G
		{Check: "all_wps_complete", Required: true},
		{Check: "all_findings_terminal", Required: true},
		{Check: "no_internal_deferrals", Required: true},
		{Check: "dod_complete", Required: true},
		{Check: "tests_pass", Required: true, EvidenceRequired: true},
	},
	"acceptance_exit": {
		{Check: "all_findings_terminal", Required: true},
		{Check: "dod_complete", Required: true},
		{Check: "no_inadequate_verdicts", Required: true},
		{Check: "deployment_verified", Required: true, EvidenceRequired: true},
	},
}

// Simple checks: computed from the graph, no evidence involved.
var simpleChecks = map[string]func(ctx context.Context, s *Store, respID string) (bool, string){
	"has_specification": func(ctx context.Context, s *Store, id string) (bool, string) {
		edges, _ := s.Neighbors(ctx, id, "out", EdgeDocumentedBy)
		if len(edges) == 0 {
			return false, "no DOCUMENTED_BY edge to a specification"
		}
		return true, ""
	},
	"work_packages_registered": func(ctx context.Context, s *Store, id string) (bool, string) {
		edges, _ := s.Neighbors(ctx, id, "in", EdgeBelongsTo)
		if len(edges) == 0 {
			return false, "no work packages BELONGS_TO this responsibility"
		}
		return true, ""
	},
	"all_wps_complete": func(ctx context.Context, s *Store, id string) (bool, string) {
		edges, _ := s.Neighbors(ctx, id, "in", EdgeBelongsTo)
		for _, e := range edges {
			wp, err := s.GetNode(ctx, e.Src)
			if err != nil || wp.Status != "COMPLETE" {
				return false, e.Src + " is not COMPLETE"
			}
		}
		return true, ""
	},
	"all_findings_terminal": func(ctx context.Context, s *Store, id string) (bool, string) {
		terminal := map[string]bool{"RESOLVED": true, "CONFIRMED": true,
			"REJECTED": true, "DEFERRED": true}
		edges, _ := s.Neighbors(ctx, id, "in", EdgeAffects)
		for _, e := range edges {
			f, err := s.GetNode(ctx, e.Src)
			if err == nil && f.Type == "Finding" && !terminal[f.Status] {
				return false, e.Src + " is " + f.Status
			}
		}
		return true, ""
	},
	"no_internal_deferrals": func(ctx context.Context, s *Store, id string) (bool, string) {
		edges, _ := s.Neighbors(ctx, id, "in", EdgeAffects)
		for _, e := range edges {
			f, err := s.GetNode(ctx, e.Src)
			if err == nil && f.Type == "Finding" && f.Status == "DEFERRED" {
				if ok, _ := s.CheckDeferral(ctx, f.ID); !ok {
					return false, f.ID + " is DEFERRED with no ExternalDependency BLOCKS edge"
				}
			}
		}
		return true, ""
	},
	// dod_complete and no_inadequate_verdicts land in 6.2 / 6.3.
}

func (s *Store) CheckGate(ctx context.Context, respID, gate string) (GateResult, error) {
	conds, ok := GateDefinitions[gate]
	if !ok {
		return GateResult{}, fmt.Errorf("unknown_gate: %q", gate)
	}
	res := GateResult{Gate: gate, Passed: true, Results: map[string]string{}}
	for _, c := range conds {
		var pass bool
		var reason string
		if c.EvidenceRequired {
			pass, reason = s.latestEvidencePasses(ctx, respID, gate, c.Check)
		} else if fn, ok := simpleChecks[c.Check]; ok {
			pass, reason = fn(ctx, s, respID)
		} else {
			pass, reason = false, "no checker registered for "+c.Check
		}
		if pass {
			res.Results[c.Check] = "pass"
		} else {
			res.Results[c.Check] = reason
			if c.Required {
				res.Passed = false
			}
		}
	}
	return res, nil
}

    Two doctrines encoded here: a check with no registered checker fails (never "passed" because nothing ran — "passed" must be distinguishable from "did not run"), and non-required checks report without blocking.

  2. Stub latestEvidencePasses to return false, "no evidence submitted" for now — 6.2 fills it in. Register the missing simple checks as failing placeholders too.

  3. Expose check_gate(responsibility_id, gate) as a tool and run it against booking-core: design_exit should pass (spec + WPs exist from week 5, findings resolved from week 3); build_exit should fail listing exactly which checks and why. A failing gate that names its reasons is a to-do list, not a wall.

Checkpoint

6.2Evidence & the Definition of Donesubmit_gate_evidence▶

"Tests pass" is a claim; a GateEvidence node with the test output attached is a fact someone can audit. Evidence is only accepted for (gate, check) pairs the gate actually declares — you can't invent a check and satisfy it. The Definition of Done is the same idea at item level: a fixed checklist of booleans, flipped one at a time, with progress reported as N/M; partial delivery is never accepted.

In the real system

submit_gate_evidence validates the pair against the GateDefinition, caps content at 10KB, and runs per-check content validators (the patterns_reviewed check requires structured JSON with resolvable pattern UUIDs; an empty list requires the literal sentinel "no applicable patterns" — silent skips are impossible). DoD: 11 items per domain responsibility, 12 with ops (WOW-007); update_dod flips one canonical key and returns "N/12". As defense-in-depth, when a runtime checker exists for the same check, both must pass — self-submitted evidence is never sole authority.

Tasks

  1. Implement evidence submission + lookup:

    server/internal/graph/evidence.go
    package graph

import (
	"context"
	"fmt"

	"github.com/google/uuid"
	"github.com/neo4j/neo4j-go-driver/v5/neo4j"
)

func (s *Store) SubmitGateEvidence(ctx context.Context, respID, gate, check, content string, pass bool) (string, error) {
	conds, ok := GateDefinitions[gate]
	if !ok {
		return "", fmt.Errorf("unknown_gate: %q", gate)
	}
	declared := false
	for _, c := range conds {
		if c.Check == check && c.EvidenceRequired {
			declared = true
		}
	}
	if !declared {
		return "", fmt.Errorf("check_not_evidence_bearing: %s does not declare "+
			"%q as evidence-required — you cannot invent a check and satisfy it",
			gate, check)
	}
	if len(content) < 20 || len(content) > 10<<10 {
		return "", fmt.Errorf("evidence_content: 20 bytes to 10KB — paste the "+
			"actual output, not an essay and not 'done'")
	}
	id := "evidence-" + uuid.NewString()
	err := s.CreateNode(ctx, Node{ID: id, Type: "GateEvidence",
		Props: map[string]any{
			"responsibility_id": respID, "gate": gate, "check": check,
			"content": content, "pass": pass,
		}})
	return id, err
}

// latestEvidencePasses: the most recent evidence node for the tuple
// decides. No evidence = fail (never "no news is good news").
func (s *Store) latestEvidencePasses(ctx context.Context, respID, gate, check string) (bool, string) {
	res, err := s.run(ctx, `
		MATCH (e:GateEvidence {project_id: $pid})
		WHERE e.responsibility_id = $rid AND e.gate = $gate AND e.check = $check
		RETURN e.pass AS pass
		ORDER BY e.created_at DESC, e.id DESC LIMIT 1`, map[string]any{
		"rid": respID, "gate": gate, "check": check,
	})
	if err != nil {
		return false, "evidence lookup failed: " + err.Error()
	}
	if len(res.Records) == 0 {
		return false, "no evidence submitted for " + check
	}
	pass, _, _ := neo4j.GetRecordValue[bool](res.Records[0], "pass")
	if pass {
		return true, ""
	}
	return false, "latest evidence for " + check + " is failing"
}

Why it exists

The sentinel-phrase trick ("no applicable patterns" required verbatim when the list is empty) exists because agents optimizing for gate-green will otherwise submit an empty list and move on — indistinguishable from having actually checked. Forcing an explicit claim converts a silent skip into an auditable lie, and auditable lies get caught.

6.3Verdicts: the reviewer ensemblerecord_verdict▶

DDE never trusts one reviewer, because each lens catches a different failure class: the skeptic hunts failure modes and error handling; consistency checks contract conformance and shared-code reuse; the craftsman judges readability and structure; the strategic reviewer asks "will we regret this in 3 sprints?". Each records a Verdict node, and gates require both presence (verdict_exists:skeptic) and quality (no_inadequate_verdicts — judged on each reviewer's latest verdict, so a re-review that comes back clean actually unblocks).

In the real system

record_verdict.go: Verdict node + VERDICT_FOR edge; canonical reviewer enum (skeptic, buildability, strategic, consistency, completeness, deployment, decision-authority…, with legacy aliases normalized on input); verdict scales differ by reviewer — strategic uses ALIGNED / MINOR_RISK / SIGNIFICANT_REGRET_RISK, everyone else ACCEPTABLE / NEEDS_WORK / INADEQUATE. build_C_to_D also requires finding_count_plausible:3 — fewer than 3 findings on a >500-LOC build fails the gate as suspiciously superficial.

6.4Hooks: enforcement that can't be forgottenexit 2▶

The final plane: mechanism. Claude Code hooks are scripts that fire around tool calls — a PreToolUse hook can block the call before it happens (exit 2, stderr becomes the agent's remediation message); a PostToolUse hook observes results. The flagship pattern you'll clone: when check_gate passes, a PostToolUse hook drops a flag file; a PreToolUse hook on update_status refuses gate-protected transitions unless the flag exists. The agent can dispatch work however it likes — it cannot claim completion without the graph's proof.

In the real system

The harness contract: one JSON object on stdin (tool_name, tool_input, and for PostToolUse tool_response); exit 0 allows, exit 2 blocks with stderr fed to the model. enforce-gate-before-status.sh is the two-mode flag-file handshake (COMPLETE requires build_phase_G or acceptance_exit by domain). Load-bearing quirk: PostToolUse fires only on exact tool-name matchers for MCP tools — the prefix matcher is inert (Finding-posttooluse-mcp-dde-WQKV). Git side: ensure-git-hooks.sh self-installs core.hooksPath=.githooks every session; pre-commit is a sectioned walk with a PASS/FAIL tally.

Tasks

  1. Write the two-mode gate hook (requires jq):

    .claude/hooks/enforce-gate-before-status.sh
    #!/usr/bin/env bash
# Two modes, wired to two events in settings.json:
#   PostToolUse on mcp__minidde__check_gate  → mint a flag on pass
#   PreToolUse  on mcp__minidde__update_status → block gated statuses
set -uo pipefail
cd "$(git rev-parse --show-toplevel 2>/dev/null || pwd)"
FLAGDIR="${HOME}/.minidde/gate-flags"; mkdir -p "$FLAGDIR"
INPUT=$(cat)
TOOL=$(jq -r '.tool_name // ""' <<<"$INPUT")

if [[ "$TOOL" == "mcp__minidde__check_gate" ]]; then
  # PostToolUse: on a passing result, mint the flag.
  RESP=$(jq -r '.tool_input.responsibility_id // ""' <<<"$INPUT")
  GATE=$(jq -r '.tool_input.gate // ""' <<<"$INPUT")
  PASSED=$(jq -r '.tool_response.content[0].text // "{}"' <<<"$INPUT" \
           | jq -r '.passed // false' 2>/dev/null)
  if [[ "$PASSED" == "true" && -n "$RESP" && -n "$GATE" ]]; then
    touch "$FLAGDIR/gate-${RESP}-${GATE}"
  fi
  exit 0
fi

if [[ "$TOOL" == "mcp__minidde__update_status" ]]; then
  TO=$(jq -r '.tool_input.to // ""' <<<"$INPUT")
  ID=$(jq -r '.tool_input.id // ""' <<<"$INPUT")
  declare -A REQUIRED=(
    [DESIGN_COMPLETE]="design_exit"
    [BUILT]="build_exit"
    [COMPLETE]="acceptance_exit"
  )
  GATE="${REQUIRED[$TO]:-}"
  [[ -z "$GATE" ]] && exit 0   # entry statuses pass freely
  if [[ ! -f "$FLAGDIR/gate-${ID}-${GATE}" ]]; then
    cat >&2 <<EOF
BLOCKED: $ID → $TO requires a passing '$GATE' gate check first.
Remedy: submit any missing evidence (submit_gate_evidence), then run
check_gate(responsibility_id="$ID", gate="$GATE"). When it passes,
retry this transition.
EOF
    exit 2
  fi
  rm -f "$FLAGDIR/gate-${ID}-${GATE}"   # flags are single-use
  exit 0
fi
exit 0

  3. Build the git tier. A sectioned pre-commit + a self-installer:

    .githooks/pre-commit
    #!/usr/bin/env bash
# Sectioned gate: each section prints PASS/FAIL; any FAIL blocks.
set -uo pipefail
fails=0
section() { echo "── $1"; }
result() { if [[ $2 -eq 0 ]]; then echo "   PASS"; else echo "   FAIL"; fails=$((fails+1)); fi; }

section "1. branch policy — no direct commits to main"
branch=$(git symbolic-ref --short HEAD 2>/dev/null || echo detached)
[[ "$branch" != "main" ]]; result "branch" $?

section "2. governance lint (decision records)"
./scripts/lint-decisions.sh >/dev/null; result "lint-decisions" $?

section "3. go build + vet (server module)"
(cd server && go build ./... && go vet ./...) >/dev/null; result "go" $?

echo; echo "pre-commit: $fails failure(s)"
exit $((fails > 0))
